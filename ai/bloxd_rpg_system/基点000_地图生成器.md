# 基点 (0,0,0) 地图生成器

以 (0, 0, 0) 为基点生成地图的代码模板。

---

## 模板1：简单建筑（5x5x5 小屋）

```javascript
// 基点: (0,0,0)
// 生成一个简单的小屋

var blueprint = [
    // ========== 地板 (Y=0) ==========
    // 5x5 石头地板
    [0,0,0,"Stone"],[1,0,0,"Stone"],[2,0,0,"Stone"],[3,0,0,"Stone"],[4,0,0,"Stone"],
    [0,0,1,"Stone"],[1,0,1,"Stone"],[2,0,1,"Stone"],[3,0,1,"Stone"],[4,0,1,"Stone"],
    [0,0,2,"Stone"],[1,0,2,"Stone"],[2,0,2,"Stone"],[3,0,2,"Stone"],[4,0,2,"Stone"],
    [0,0,3,"Stone"],[1,0,3,"Stone"],[2,0,3,"Stone"],[3,0,3,"Stone"],[4,0,3,"Stone"],
    [0,0,4,"Stone"],[1,0,4,"Stone"],[2,0,4,"Stone"],[3,0,4,"Stone"],[4,0,4,"Stone"],
    
    // ========== 墙壁 (Y=1~3) ==========
    // 四角柱子
    [0,1,0,"Wood"],[4,1,0,"Wood"],[0,1,4,"Wood"],[4,1,4,"Wood"],
    [0,2,0,"Wood"],[4,2,0,"Wood"],[0,2,4,"Wood"],[4,2,4,"Wood"],
    [0,3,0,"Wood"],[4,3,0,"Wood"],[0,3,4,"Wood"],[4,3,4,"Wood"],
    
    // 门框
    [2,1,0,"Air"],[2,2,0,"Air"],
    
    // ========== 屋顶 (Y=4) ==========
    [0,4,0,"Cobblestone"],[1,4,0,"Cobblestone"],[2,4,0,"Cobblestone"],[3,4,0,"Cobblestone"],[4,4,0,"Cobblestone"],
    [0,4,1,"Cobblestone"],[1,4,1,"Cobblestone"],[2,4,1,"Cobblestone"],[3,4,1,"Cobblestone"],[4,4,1,"Cobblestone"],
    [0,4,2,"Cobblestone"],[1,4,2,"Cobblestone"],[2,4,2,"Cobblestone"],[3,4,2,"Cobblestone"],[4,4,2,"Cobblestone"],
    [0,4,3,"Cobblestone"],[1,4,3,"Cobblestone"],[2,4,3,"Cobblestone"],[3,4,3,"Cobblestone"],[4,4,3,"Cobblestone"],
    [0,4,4,"Cobblestone"],[1,4,4,"Cobblestone"],[2,4,4,"Cobblestone"],[3,4,4,"Cobblestone"],[4,4,4,"Cobblestone"],
    
    // ========== 内部装饰 ==========
    // 中心信标
    [2,1,2,"Beacon"],
    // 火把
    [1,2,1,"Torch"],[3,2,1,"Torch"],[1,2,3,"Torch"],[3,2,3,"Torch"],
];

// 放置所有方块
for(var i=0;i<blueprint.length;i++){
    var b=blueprint[i];
    if(b[3]!=="Air"){
        api.setBlock(b[0],b[1],b[2],b[3]);
    }
}

api.broadcastMessage([{str:"[蓝图] (0,0,0)基点小屋已生成!",style:{color:"green",fontWeight:"bold"}}]);
```

## 模板2：修仙大厅（3x3 区域标记）

```javascript
// 基点: (0,0,0)
// 3x3 修仙大厅 - 仅标记

var areas = [
    // 第一行 (Z=-80)
    {x:-80,z:-80,color:"Red Concrete",name:"炼气区"},
    {x:0,z:-80,color:"Green Concrete",name:"筑基区"},
    {x:80,z:-80,color:"Blue Concrete",name:"金丹区"},
    
    // 第二行 (Z=0) - 中间是基点
    {x:-80,z:0,color:"Yellow Concrete",name:"元婴区"},
    {x:0,z:0,color:"Moonstone",name:"出生点-基点",isSpawn:true},
    {x:80,z:0,color:"Purple Concrete",name:"化神区"},
    
    // 第三行 (Z=80)
    {x:-80,z:80,color:"Cyan Concrete",name:"炼虚区"},
    {x:0,z:80,color:"Orange Concrete",name:"合体区"},
    {x:80,z:80,color:"Pink Concrete",name:"大乘区"}
];

for(var i=0;i<areas.length;i++){
    var a=areas[i];
    // 5x5底座
    for(var dx=-2;dx<=2;dx++){
        for(var dz=-2;dz<=2;dz++){
            api.setBlock(a.x+dx,0,a.z+dz,a.color);
        }
    }
    // 中心信标
    api.setBlock(a.x,1,a.z,"Beacon");
    
    // 出生点特殊处理
    if(a.isSpawn){
        for(var dx=-3;dx<=3;dx++){
            for(var dz=-3;dz<=3;dz++){
                if(Math.abs(dx)>2||Math.abs(dz)>2){
                    api.setBlock(a.x+dx,0,a.z+dz,"White Concrete");
                }
            }
        }
    }
}

api.broadcastMessage([{str:"[蓝图] (0,0,0)基点3x3大厅已生成!",style:{color:"green",fontWeight:"bold"}}]);
```

## 模板3：空白地图（仅供坐标参考）

```javascript
// 基点: (0,0,0)
// 空白地图 - 用于测试坐标

// 在基点放置标记
api.setBlock(0,0,0,"Red Concrete");
api.setBlock(0,1,0,"Beacon");

// X轴标记（每10格）
for(var x=-50;x<=50;x+=10){
    if(x!==0){
        api.setBlock(x,0,0,"Stone");
    }
}

// Z轴标记（每10格）
for(var z=-50;z<=50;z+=10){
    if(z!==0){
        api.setBlock(0,0,z,"Stone");
    }
}

// 四角标记
api.setBlock(50,0,50,"Yellow Concrete");
api.setBlock(-50,0,50,"Yellow Concrete");
api.setBlock(50,0,-50,"Yellow Concrete");
api.setBlock(-50,0,-50,"Yellow Concrete");

api.broadcastMessage([{str:"[蓝图] (0,0,0)基点坐标参考已生成! X/Z范围: -50~50",style:{color:"green"}}]);
```

## 自定义模板

如果你有具体的方块坐标，使用这个格式：

```javascript
// 基点: (0,0,0)
// 格式: [x, y, z, "方块名"]

var blueprint = [
    // 在这里添加你的方块坐标
    // 例如:
    [0,0,0,"Stone"],
    [1,0,0,"Stone"],
    [2,0,0,"Stone"],
    // ...
];

// 放置代码
for(var i=0;i<blueprint.length;i++){
    var b=blueprint[i];
    if(b[3]!=="Air"){
        api.setBlock(b[0],b[1],b[2],b[3]);
    }
}

api.broadcastMessage([{str:"[蓝图] 自定义地图已生成!",style:{color:"green"}}]);
```

---

## 坐标说明

- **基点 (0,0,0)**: 地图的中心/起始点
- **X轴**: 东西方向（东+，西-）
- **Y轴**: 上下高度（上+，下-）
- **Z轴**: 南北方向（南+，北-）

| Torch | 火把 |
| Water | 水 |
| Lava | 岩浆 |
