- **炼虚区**: 东南 (50,-50)
- **合体区**: 正东 (50,0)
- **大乘区**: 东北 (50,50)

## 地图生成代码（3x3大厅）

```javascript
// 基点: (0,0,0) 为中心
// 生成3x3修仙大厅

var areas = [
    // 第一行 (Z=-50)
    {x:-50,z:-50,color:"Red Concrete",name:"炼气区"},
    {x:0,z:-50,color:"Blue Concrete",name:"元婴区"},
    {x:50,z:-50,color:"Cyan Concrete",name:"炼虚区"},
    
    // 第二行 (Z=0) - 中间是基点
    {x:-50,z:0,color:"Green Concrete",name:"筑基区"},
    {x:0,z:0,color:"Moonstone",name:"出生点-基点",isSpawn:true},
    {x:50,z:0,color:"Purple Concrete",name:"合体区"},
    
    // 第三行 (Z=50)
    {x:-50,z:50,color:"Yellow Concrete",name:"金丹区"},
    {x:0,z:50,color:"Gray Concrete",name:"化神区"},
    {x:50,z:50,color:"Pink Concrete",name:"大乘区"}
];

for(var i=0;i<areas.length;i++){
    var a=areas[i];
    // 7x7底座
    for(var dx=-3;dx<=3;dx++){
        for(var dz=-3;dz<=3;dz++){
            api.setBlock(a.x+dx,0,a.z+dz,a.color);
        }
    }
    // 中心信标
    api.setBlock(a.x,1,a.z,"Beacon");
    
    // 出生点特殊处理
    if(a.isSpawn){
        for(var dx=-4;dx<=4;dx++){
            for(var dz=-4;dz<=4;dz++){
                if(Math.abs(dx)>3||Math.abs(dz)>3){
                    api.setBlock(a.x+dx,0,a.z+dz,"White Concrete");
                }
            }
        }
    }
}

api.broadcastMessage([{str:"[蓝图] (0,0,0)基点3x3修仙大厅已生成!",style:{color:"green",fontWeight:"bold"}}]);
```

## 使用步骤

1. **放置世界代码**到Bloxd世界代码区域
2. **放置配置代码**到代码块（放在世界代码之后）
3. **放置地图生成代码**到代码块（可选）
4. 玩家出生在 (0,0,0) 基点
5. 走向不同方向的修炼区域开始刷怪修炼
