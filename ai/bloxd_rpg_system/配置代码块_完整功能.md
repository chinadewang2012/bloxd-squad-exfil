## 基础配置（点击加载）

```javascript
setConfig({
    title: "修仙世界",
    // 数字格式化函数
    fmtNum: function(n) {
        var u = [[1e68, "大数"], [1e64, "不可思议"], [1e60, "那由他"], [1e56, "阿僧祇"], 
                 [1e52, "恒河沙"], [1e48, "极"], [1e44, "载"], [1e40, "正"], [1e36, "涧"], 
                 [1e32, "沟"], [1e28, "穰"], [1e24, "秭"], [1e20, "垓"], [1e16, "京"], 
                 [1e12, "兆"], [1e8, "亿"], [1e4, "万"]];
        for (var i = 0; i < u.length; i++) {
            if (n >= u[i][0]) return (n / u[i][0]).toFixed(1) + u[i][1];
        }
        return "" + n;
    },
    // 战力计算函数
    getPower: function(st) {
        return Math.round((st.hp + st.atk) * 10 * 100) / 100;
    },
    // 境界获取函数
    getRealm: function(pw) {
        var rn = "凡人", rc = "#888888";
        var jj = [[1e5, "炼气", "#88dddd"], [1e7, "筑基", "#90ee90"], [1e8, "金丹", "#ffee88"], 
                  [1e9, "元婴", "#cc99ff"], [1e11, "化神", "#ff0000"], [5e11, "炼虚", "#cccccc"], 
                  [1e13, "合体", "#aa88ff"], [3e14, "大乘", "#ffccdd"], [1e38, "真仙", "#ffffff"], 
                  [9e73, "硅基", "#ffdd44"]];
        for (var i = 0; i < jj.length; i++) {
            if (pw >= jj[i][0]) {
                rn = jj[i][1];
                rc = jj[i][2];
            }
        }
        return {name: rn, color: rc};
    },
    // 刷怪区域
    zones: {
        lianqi: {
            name: "炼气修炼场",
            zone: [-50, -30, 0, 20, -50, -30],
            rx: [-48, -32],
            ry: [2, 10],
            rz: -40,
            mob: "Zombie",
            mobName: "炼气僵尸",
            max: 5,
            count: 3,
            dropRate: 0.4,
            dan: [
                {name: "聚气丹", item: "Apple", hp: 100, atk: 10},
                {name: "练气散", item: "Melon Slice", hp: 300, atk: 30}
            ]
        }
    }
});

// 放置区域标记
api.setBlock(-40, 0, -40, "Red Concrete");
for (var dx = -2; dx <= 2; dx++) {
    for (var dz = -2; dz <= 2; dz++) {
        api.setBlock(-40 + dx, 0, -40 + dz, "Red Concrete");
    }
}
api.setBlock(-40, 1, -40, "Beacon");

api.broadcastMessage([{str: "[地图] 炼气区已生成!请到 (-40, -40) 修炼!", style: {color: "green", fontWeight: "bold"}}]);
```

## 使用说明

1. **先粘贴**【极简世界代码】到代码块并激活
2. **再点击**上方任意【配置代码块】加载功能
3. 看到绿色提示后即可开始修炼

## 配置内容

| 配置项 | 说明 |
|--------|------|
| `fmtNum` | 数字格式化函数（万、亿、兆等） |
| `getPower` | 战力计算公式 |
| `getRealm` | 境界判断逻辑 |
| `zones` | 刷怪区域配置 |
