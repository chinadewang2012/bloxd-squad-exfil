# Bloxd修仙RPG - 完整可配置版（世界代码）

```javascript
// ========== 基础变量 ==========
_zIds=[];_zMax=5;_inZone={};_tC=0;_cfg={};_init=false;_zones={};

// ========== 工具函数 ==========
_rd=function(s){if(!s||!s.attributes||!s.attributes.customDescription)return 0;var m=s.attributes.customDescription.match(/[0-9.eE+]+/);return m?parseFloat(m[0]):0};
_getStats=function(p){var s0=api.getMoonstoneChestItemSlot(p,0),s1=api.getMoonstoneChestItemSlot(p,1),s2=api.getMoonstoneChestItemSlot(p,2),s3=api.getMoonstoneChestItemSlot(p,3),s4=api.getMoonstoneChestItemSlot(p,4);return{atk:_rd(s0)||10,hp:_rd(s1)||10,reborn:_rd(s2)||0,mult:_rd(s3)||1,jade:_rd(s4)||0}};
_setStats=function(p,atk,hp,reborn,mult,jade){atk=Math.round(atk*100)/100;hp=Math.round(hp*100)/100;api.setMoonstoneChestItemSlot(p,0,"Red Concrete",1,{customDisplayName:"攻击",customDescription:""+atk});api.setMoonstoneChestItemSlot(p,1,"Green Concrete",1,{customDisplayName:"生命",customDescription:""+hp});api.setMoonstoneChestItemSlot(p,2,"Cyan Concrete",1,{customDisplayName:"轮回",customDescription:""+reborn});if(mult!==undefined)api.setMoonstoneChestItemSlot(p,3,"Yellow Concrete",1,{customDisplayName:"倍率",customDescription:""+mult});if(jade!==undefined)api.setMoonstoneChestItemSlot(p,4,"Blue Concrete",1,{customDisplayName:"仙玉",customDescription:""+jade})};
_fmtNum=function(n){var u=[[1e68,"大数"],[1e64,"不可思议"],[1e60,"那由他"],[1e56,"阿僧祇"],[1e52,"恒河沙"],[1e48,"极"],[1e44,"载"],[1e40,"正"],[1e36,"涧"],[1e32,"沟"],[1e28,"穰"],[1e24,"秭"],[1e20,"垓"],[1e16,"京"],[1e12,"兆"],[1e8,"亿"],[1e4,"万"]];for(var i=0;i<u.length;i++)if(n>=u[i][0])return(n/u[i][0]).toFixed(1)+u[i][1];return""+n};
_getPower=function(st){return Math.round((st.hp+st.atk)*10*100)/100};
_getRealm=function(pw){var rn="凡人",rc="#888888";var jj=_cfg.jj||[[1e5,"炼气","#88dddd"],[1e7,"筑基","#90ee90"],[1e8,"金丹","#ffee88"],[1e9,"元婴","#cc99ff"],[1e11,"化神","#ff0000"],[5e11,"炼虚","#cccccc"],[1e13,"合体","#aa88ff"],[3e14,"大乘","#ffccdd"],[1e38,"真仙","#ffffff"],[9e73,"硅基","#ffdd44"]];for(var i=0;i<jj.length;i++)if(pw>=jj[i][0]){rn=jj[i][1];rc=jj[i][2]}return{name:rn,color:rc}};
_updSB=function(p){var n=api.getEntityName(p),st=_getStats(p),pw=_getPower(st),rl=_getRealm(pw);api.setClientOption(p,"RightInfoText",[{str:"══"+(_cfg.title||"修仙")+"══",style:{color:"gold",fontSize:"18px",fontWeight:"bold"}},{str:n,style:{color:"white",fontSize:"15px"}},{str:"境界:"+rl.name,style:{color:rl.color,fontSize:"14px",fontWeight:"bold"}},{str:"战力:"+_fmtNum(pw),style:{color:"red",fontSize:"14px"}},{str:"攻击:"+_fmtNum(st.atk),style:{color:"orange",fontSize:"14px"}},{str:"生命:"+_fmtNum(st.hp),style:{color:"lime",fontSize:"14px"}},{str:"轮回:"+st.reborn,style:{color:"cyan",fontSize:"14px"}},{str:"倍率:x"+_fmtNum(st.mult),style:{color:"yellow",fontSize:"14px"}},{str:"仙玉:"+_fmtNum(st.jade),style:{color:"#00ddff",fontSize:"14px"}}])};
_clrZ=function(ids){for(var i=0;i<ids.length;i++)try{api.despawnMob(ids[i].id)}catch(e){}ids.length=0};
_spawnMobAt=function(zn,px,py){var z=_zones[zn];if(!z||z.ids.length>=z.max)return;var rx=z.rx[0]+Math.floor(Math.random()*(z.rx[1]-z.rx[0]));var ry=z.ry[0]+Math.floor(Math.random()*(z.ry[1]-z.ry[0]));var rz=z.rz;if(Math.abs(rx-px)<2&&Math.abs(ry-py)<2)rx=px+5;var id=api.attemptSpawnMob(z.mob,rx,ry,rz,{name:z.mobName});if(id)z.ids.push({id:id,time:Date.now()})};
_sp=function(zn,px,py){var z=_zones[zn];for(var i=0;i<z.count;i++)_spawnMobAt(zn,px,py)};

// ========== 官方回调 ==========
tick=function(ms){if(!_init)return;_tC++;for(var i=0;i<api.getPlayerIds().length;i++){var p=api.getPlayerIds()[i];if(!api.playerIsInGame(p))continue;if(!_inZone[p])_inZone[p]={};var pos=api.getPosition(p);for(var zn in _zones){var z=_zones[zn];var near=pos&&pos[0]>=z.zone[0]&&pos[0]<=z.zone[1]&&pos[1]>=z.zone[2]&&pos[1]<=z.zone[3]&&pos[2]>=z.zone[4]&&pos[2]<=z.zone[5];if(near){if(!_inZone[p][zn]){_inZone[p][zn]=true;api.sendFlyingMiddleMessage(p,["进入"+z.name+"!"],100,3000);_sp(zn,pos[0],pos[1])}else if(_tC%100===0)_sp(zn,pos[0],pos[1])}else{if(_inZone[p][zn]){_inZone[p][zn]=false;api.sendFlyingMiddleMessage(p,["离开"+z.name+"!"],100,2000);_clrZ(z.ids)}}}}if(_tC%200===0)for(var j=0;j<api.getPlayerIds().length;j++){var pid=api.getPlayerIds()[j];if(api.playerIsInGame(pid))_updSB(pid)}};

onPlayerJoin=function(p,f){var st=_getStats(p);if(st.atk<=10)_setStats(p,10,10,0,1,0);_inZone[p]={};_updSB(p);api.setHealth(p,Math.floor(st.hp),undefined,true)};
onPlayerLeave=function(p,s){_inZone[p]={}};
onPlayerChat=function(p,m,c){var st=_getStats(p),pw=_getPower(st),rl=_getRealm(pw),n=api.getEntityName(p);api.broadcastMessage([{str:"[",style:{color:"gray"}},{str:rl.name,style:{color:rl.color,fontWeight:"bold"}},{str:"] ",style:{color:"gray"}},{str:n+": ",style:{color:"white"}},{str:m,style:{color:"#cccccc"}}]);return false};
onPlayerKilledMob=function(p,mb,d,w){if(!_init)return;var pos=api.getPosition(mb);if(!pos)return;var zn=null;for(var k in _zones){if(_inZone[p]&&_inZone[p][k]){zn=k;break}}if(!zn)return;var z=_zones[zn];if(Math.random()<(z.dropRate||0.3)){var dan=z.dan[Math.floor(Math.random()*z.dan.length)];api.createItemDrop(pos[0],pos[1],pos[2],dan.item,1,false,{customDisplayName:dan.name,customDescription:"生命+"+dan.hp+" 攻击+"+dan.atk},30000,null,{doPhysics:true})}};
onPlayerDamagingOtherPlayer=function(ap,dp,d,w,b,db){var st=_getStats(ap);return d*st.mult};
onPlayerDamagingMob=function(p,mb,d,w,db){var st=_getStats(p);return d*st.mult};
onPlayerAttemptAltAction=function(p,x,y,z,bl,t){if(!_init||!t)return;var item=api.getHeldItem(p);if(!item||!item.customDisplayName)return;var dan=null;for(var zn in _zones){for(var i=0;i<_zones[zn].dan.length;i++){if(_zones[zn].dan[i].name===item.customDisplayName){dan=_zones[zn].dan[i];break}}if(dan)break}if(dan){var st=_getStats(p);_setStats(p,st.atk+dan.atk,st.hp+dan.hp,st.reborn,st.mult,st.jade);_updSB(p);api.sendFlyingMiddleMessage(p,["使用了 "+dan.name+"! 生命+"+dan.hp+" 攻击+"+dan.atk],100,2000);api.removeItemName(p,item.name,1);return"preventAction"}};

// ========== 配置接口 ==========
setConfig=function(c){_cfg=c;_init=true;if(c.zones){for(var zn in c.zones){_zones[zn]=c.zones[zn];_zones[zn].ids=[]}}api.broadcastMessage([{str:"[系统] "+(c.title||"修仙")+"系统已加载!",style:{color:"green",fontWeight:"bold"}}])};
addZone=function(zn,z){if(!_zones[zn]){_zones[zn]=z;_zones[zn].ids=[];return true}return false};
removeZone=function(zn){if(_zones[zn]){_clrZ(_zones[zn].ids);delete _zones[zn];return true}return false};
```

## 配置接口说明

### setConfig(config) - 加载完整配置
设置整个游戏系统的配置。

### addZone(zoneName, zoneConfig) - 添加区域
动态添加新的刷怪区域。

### removeZone(zoneName) - 移除区域
动态移除刷怪区域。
